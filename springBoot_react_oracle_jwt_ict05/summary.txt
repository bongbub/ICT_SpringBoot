--------------------[ application.properties 작성 ] ----------------------------------

springBoot_basic  프로젝트의 application.properties 파일 복사해서, 필요한 부분만 남기고 제거하거나 수정

>>>
spring.application.name=springBoot_react_oracle_jwt_ict05


# db connection(oracle) - spring(context.xml)
spring.datasource.driver-class-name=oracle.jdbc.driver.OracleDriver
spring.datasource.url=jdbc:oracle:thin:@localhost:1521:xe
spring.datasource.username=springboot_ict05
spring.datasource.password=tiger

# http port number(default 8080)
server.port=8081

# JPA SQL SHOW
spring.jpa.database-platform=org.hibernate.dialect.Oracle12cDialect
spring.jpa.show-sql=true



# custom error page
server.error.whitelabel.enabled=false

# MyBatis - spring(dataSource-config.xml)
mybatis.mapper-locations=mappers/**/*.xml
mybatis.type-aliases-package=fullstack.jwt.back_end.dao
# 주의 : false를 주면 강제변환을 막는다. 강제변환 : bNum -> b_num,  b_num -> b_num, 예외) nim -> num 한단어인 경우는 true이어도 변환안함
mybatis.configuration.map-underscore-to-camel-case=false
logging.level.com.atoz_develop.mybatissample.repository=TRACE
